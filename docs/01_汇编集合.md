# 汇编集合

本清单展示 ROM 程序的可读汇编形式。实际保存在
`rtl/mcu4_worker_instr_rom.vhd` 中的 32-bit 指令字采用
`Armv7-A/R AArch32 A32` 标准编码子集；每类指令的 bit 字段见
`docs/02_指令定义.md`。

## 记号

```text
[r0, #0x40 + 4*n] = unified dmem word[16+n]，当前程序工作区 A
[r0, #0x80 + 4*n] = unified dmem word[32+n]，当前程序工作区 B/输出区
NOP                = MOV r0, r0 = 0xE1A00000
B .                = 0xEAFFFFFE，标准 A32 自跳转，作为顶层完成哨兵
```

## FFT worker0

```asm
pc00: MOV     r0, #0
pc01: MOV     r1, #91
pc02: PKHBT   r3, r1, r1, LSL #16
pc03: SSUB16  r4, r0, r3

pc04: LDR     r5, [r0, #0x40]
pc05: LDR     r6, [r0, #0x44]
pc06: SADD16  r10, r5, r6
pc07: SSUB16  r11, r5, r6
pc08: STR     r10, [r0, #0x80]
pc09: STR     r11, [r0, #0x84]

pc10: LDR     r5, [r0, #0x80]
pc11: LDR     r6, [r0, #0x88]
pc12: SADD16  r10, r5, r6
pc13: SSUB16  r11, r5, r6
pc14: STR     r10, [r0, #0x40]
pc15: STR     r11, [r0, #0x48]

pc16: NOP

pc17: LDR     r5, [r0, #0x40]
pc18: LDR     r6, [r0, #0x50]
pc19: SADD16  r10, r5, r6
pc20: SSUB16  r11, r5, r6
pc21: STR     r10, [r0, #0x80]
pc22: STR     r11, [r0, #0x90]

pc23: NOP
pc24: NOP
pc25: NOP
pc26: NOP
pc27: B       .
```

## 基础指令测试

```asm
pc00: MOV     r0, #0
pc01: MOV     r1, #7
pc02: STR     r1, [r0, #0x44]   ; data[1] = MOV result = 7
pc03: MOV     r2, #3
pc04: ADD     r3, r1, r2
pc05: STR     r3, [r0, #0x48]   ; data[2] = ADD result = 10
pc06: SUB     r4, r3, r2
pc07: STR     r4, [r0, #0x4C]   ; data[3] = SUB result = 7
pc08: AND     r5, r3, r1
pc09: STR     r5, [r0, #0x50]   ; data[4] = AND result = 2
pc10: ORR     r6, r5, r2
pc11: STR     r6, [r0, #0x54]   ; data[5] = ORR result = 3
pc12: LDR     r8, [r0, #0x40]   ; data[0] input = 5
pc13: STR     r8, [r0, #0x58]   ; data[6] = LDR result = 5
pc14: MOV     r11, #0
pc15: B       pc17
pc16: MOV     r11, #99          ; skipped poison write
pc17: BL      pc21
pc18: ADD     r12, r10, r11
pc19: STR     r12, [r0, #0x5C]  ; data[7] = control-flow result = 15
pc20: B       .
pc21: MOV     r10, #15
pc22: MOV     pc, lr
```
